cmake_minimum_required(VERSION 3.22)
project(OBSClock VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(OBS_INCLUDE_DIR "" CACHE PATH "OBS 33 libobs headers")
set(OBS_GENERATED_INCLUDE_DIR "" CACHE PATH "Configured obsconfig.h directory")
set(OBS_IMPORT_LIBRARY "" CACHE FILEPATH "OBS library or Windows import library")
set(OBS_EXTRA_INCLUDE_DIR "" CACHE PATH "Additional OBS dependency headers such as SIMDe")
if(OBS_INCLUDE_DIR AND OBS_IMPORT_LIBRARY)
 add_library(obs_sdk UNKNOWN IMPORTED)
 set_target_properties(obs_sdk PROPERTIES IMPORTED_LOCATION "${OBS_IMPORT_LIBRARY}"
  INTERFACE_INCLUDE_DIRECTORIES "${OBS_INCLUDE_DIR};${OBS_GENERATED_INCLUDE_DIR};${OBS_EXTRA_INCLUDE_DIR}")
 set(OBS_CLOCK_LIBOBS obs_sdk)
else()
 find_package(libobs 33 CONFIG REQUIRED)
 set(OBS_CLOCK_LIBOBS OBS::libobs)
endif()
# Derive embedded bytes from the tracked artwork, rather than storing a second
# multi-megabyte copy of all images in Git.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(GLOB_RECURSE CLOCK_ASSETS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/data/*")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CLOCK_ASSETS})
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed_assets.py"
 RESULT_VARIABLE embed_result)
if(NOT embed_result EQUAL 0)
 message(FATAL_ERROR "Unable to embed clock artwork")
endif()
add_library(OBS-Clock MODULE src/clock.cpp)
target_link_libraries(OBS-Clock PRIVATE ${OBS_CLOCK_LIBOBS})
set_target_properties(OBS-Clock PROPERTIES PREFIX "" OUTPUT_NAME "OBS-Clock")
if(WIN32)
 enable_language(RC)
 target_sources(OBS-Clock PRIVATE resources/version.rc)
 target_include_directories(OBS-Clock PRIVATE resources)
 target_compile_definitions(OBS-Clock PRIVATE NOMINMAX)
 target_link_libraries(OBS-Clock PRIVATE gdi32 user32)
 if(MINGW)
  target_link_options(OBS-Clock PRIVATE -static -static-libgcc -static-libstdc++)
 endif()
 install(TARGETS OBS-Clock LIBRARY DESTINATION OBS-Clock RUNTIME DESTINATION OBS-Clock)
else()
 find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui)
 target_link_libraries(OBS-Clock PRIVATE Qt6::Core Qt6::Gui)
 if(APPLE)
  set_target_properties(OBS-Clock PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION plugin
   MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/resources/Info.plist.in"
   BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/../Frameworks")
  install(TARGETS OBS-Clock BUNDLE DESTINATION . LIBRARY DESTINATION .)
 else()
  include(GNUInstallDirs)
  install(TARGETS OBS-Clock LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/obs-plugins")
 endif()
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(clock_time_tests tests/time_test.cpp)
 add_test(NAME clock_time_tests COMMAND clock_time_tests)
 if(MSVC)
  target_compile_options(clock_time_tests PRIVATE /UNDEBUG)
 else()
  target_compile_options(clock_time_tests PRIVATE -UNDEBUG)
 endif()
 if(NOT WIN32)
  add_executable(clock_platform_tests tests/platform_test.cpp)
  target_include_directories(clock_platform_tests PRIVATE ${OBS_INCLUDE_DIR} ${OBS_GENERATED_INCLUDE_DIR})
  target_link_libraries(clock_platform_tests PRIVATE Qt6::Core Qt6::Gui)
  add_test(NAME clock_platform_tests COMMAND clock_platform_tests)
  set_tests_properties(clock_platform_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
 endif()
endif()
